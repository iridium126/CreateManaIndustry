cmake_minimum_required(VERSION 3.24)
project(CmiXessFg LANGUAGES CXX)
find_package(JNI REQUIRED)
set(XESS_SDK_DIR "" CACHE PATH "Intel XeSS SDK 3.0.2 directory")
find_path(XESS_INCLUDE xess_fg/xefg_swapchain_d3d12.h PATHS "${XESS_SDK_DIR}/inc" NO_DEFAULT_PATH REQUIRED)
find_library(XESS_FG_LIBRARY libxess_fg PATHS "${XESS_SDK_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(XELL_LIBRARY libxell PATHS "${XESS_SDK_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
function(configure_xessfg_target target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(${target} PRIVATE ${JNI_INCLUDE_DIRS} "${XESS_INCLUDE}")
    target_link_libraries(${target} PRIVATE d3d12 d3dcompiler dxgi user32 advapi32 "${XESS_FG_LIBRARY}" "${XELL_LIBRARY}")
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endfunction()

add_library(CmiXessFg SHARED XessFg.cpp)
configure_xessfg_target(CmiXessFg)

option(CMI_XESSFG_BUILD_PROBE "Build the RTX and Arc hardware transport probe" OFF)
if(CMI_XESSFG_BUILD_PROBE)
    enable_testing()
    add_executable(CmiXessFgPipelineProbe ../../../test/native/xessfg/PipelineProbe.cpp)
    target_include_directories(CmiXessFgPipelineProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    configure_xessfg_target(CmiXessFgPipelineProbe)
    add_test(NAME xessfg_pipeline_probe COMMAND "${CMAKE_COMMAND}" -E env
        --modify "PATH=path_list_prepend:${XESS_SDK_DIR}/bin" "$<TARGET_FILE:CmiXessFgPipelineProbe>")
    add_test(NAME xessfg_offline_probe COMMAND "${CMAKE_COMMAND}" -E env
        --modify "PATH=path_list_prepend:${XESS_SDK_DIR}/bin" "$<TARGET_FILE:CmiXessFgPipelineProbe>" --offline)
    set_tests_properties(xessfg_pipeline_probe PROPERTIES TIMEOUT 60)
endif()
