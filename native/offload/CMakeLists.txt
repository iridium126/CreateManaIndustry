cmake_minimum_required(VERSION 3.24)
project(CmiHardwareOffload LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED)
add_library(CmiHardwareOffload SHARED Queries.cpp Bridge.cpp AsyncSession.cpp MediaProbe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../.gradle/hardware-offload/media-headers/lock.json")
    target_include_directories(CmiHardwareOffload PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.gradle/hardware-offload/media-headers")
endif()
target_include_directories(CmiHardwareOffload PRIVATE ${JNI_INCLUDE_DIRS} "${CUDA_ROOT}/include" "${NVRTC_ROOT}/include" "${CRT_ROOT}/include" "${OPTIX_ROOT}/include")
target_link_libraries(CmiHardwareOffload PRIVATE "${CUDA_ROOT}/lib/x64/cuda.lib" "${NVRTC_ROOT}/lib/x64/nvrtc.lib" d3d12 dxgi opengl32)
target_compile_definitions(CmiHardwareOffload PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
